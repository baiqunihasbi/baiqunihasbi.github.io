                            <h2 class="text-2xl font-bold border-b border-navy-muted pb-2">Biography</h2>

                            <p>

                                I am a global historian and Assistant Professor specializing in <strong>Modern Muslim Society in Southeast Asia</strong>. Currently, I serve as an Assistant Professor and the <strong>Head of Research and Publication</strong> at <span class="font-semibold text-navy">Universitas Islam Negeri (UIN) Sultanah Nahrasiyah Lhokseumawe</span> in Aceh, Indonesia. My research lies at the intersection of historical narratives and contemporary global dynamics.

                            </p>

                            <p>

                                I earned my Ph.D. in Global History from the <strong>University of North Carolina at Chapel Hill (USA)</strong> in 2023. My dissertation, <em>Reforming the Unequal International Order: Three Stages of Indonesian Worldmaking Visions (1873-1950)</em>, maps how Islamic and Southeast Asian actors mobilized political and theological frameworks to reconstruct international hierarchies and self-determination concepts.

                            </p>

                            

                            <h3 class="text-xl font-bold text-navy pt-4">Research Fields & Scholarly Interests</h3>

                            <ul class="list-disc list-inside space-y-2 pl-2 text-slate-700">

                                <li><strong>Transregional Islamic Networks:</strong> Exploring historic diplomatic, cultural, and ideological connections between Southeast Asia (such as the Aceh Sultanate) and the wider Islamicate world, including the Ottoman Empire.</li>

                                <li><strong>Decolonial Praxis & Worldmaking:</strong> Re-evaluating anti-colonial struggles (such as the Acehnese <em>Perang Sabil</em>) as sophisticated intellectual endeavors seeking to build alternative international orders.</li>

                                <li><strong>Gender & Public Spaces:</strong> Exploring the socio-spatial transformations of modern Muslim environments, including the gendered dynamics of public and civic spaces in post-conflict Aceh.</li>

                            </ul>


                            <h3 class="text-xl font-bold text-navy pt-4">Democratizing Knowledge: The Instagram Project</h3>

                            <p>

                                A core tenet of my educational philosophy is that high-level scholarly works must transcend the boundaries of elite peer-reviewed journals and institutional paywalls. 

                            </p>

                            <p class="border-l-4 border-carolina pl-4 italic bg-white p-4 rounded-r-lg shadow-sm text-slate-800">

                                "History shouldn't live in ivory towers. My ongoing public scholarship initiative translates dense academic papers, historical archival records, and decolonial critiques into highly visual, digestible, and engaging slide-carousel posts on Instagram."

                            </p>

                            <p>

                                Through this project, I break down complex historiographies—such as the 19th-century geopolitics of Lord Stanley or Ottoman-Aceh secret treaties—into modern educational cards. This allows university students, history buffs, and the broader public to interact directly with history right on their feeds. 

                            </p>

        function switchTab(tabId) {

            // Hide all tab sections

            document.querySelectorAll('section[id^="page-"]').forEach(section => {

                section.classList.add('hidden');

                section.classList.remove('block');

            });

            

            // Show requested tab section

            const targetSection = document.getElementById(`page-${tabId}`);

            targetSection.classList.remove('hidden');

            targetSection.classList.add('block');


            // Reset navigation button states to inactive

            document.querySelectorAll('nav button').forEach(button => {

                button.classList.remove('text-navy', 'border-navy', 'font-bold');

                button.classList.add('border-transparent', 'text-slate-500');

            });


            // Set current tab button as active

            const activeNav = document.getElementById(`nav-${tabId}`);

            if (activeNav) {

                activeNav.classList.add('text-navy', 'border-navy', 'font-bold');

                activeNav.classList.remove('border-transparent', 'text-slate-500');

            }

            const companionBox = document.getElementById('companion-content');

            companionBox.innerHTML = blockBlueprints[tabId] || '<p class="text-xs text-slate-400">Loading companion guide...</p>';

            

            // Scroll view back to top smoothly

            window.scrollTo({ top: 0, behavior: 'smooth' });

        }
